# USA paratrooper research
EAI_AIR_USA_paratrooper_research = {
	enable = {
		tag = USA
		is_ai = yes
		date > 1940.1.1
	}
	
	abort_when_not_enabled = yes
	
	# Increase priority for paratroopers tech
	ai_strategy = {
		type = research_tech
		id = paratroopers
		value = 50
	}
	
	ai_strategy = {
		type = research_tech
		id = paratroopers2
		value = 40
	}
	
	# Ensure transport planes get researched

	}

	}
