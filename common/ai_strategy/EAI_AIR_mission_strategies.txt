# This system improves AI air force management by dynamically adjusting air mission priorities

### Dynamic Air Mission Prioritization Based on War Phase ###

# Early war phase - prioritize air superiority and reconnaissance
EAI_AIR_early_war_mission_priority = {
	enable = {
		has_war = yes
		OR = {
			AND = {
				date < 1940.1.1
				has_war_with = FROM
				NOT = { has_country_flag = EAI_late_war }
			}
			has_country_flag = { flag = EAI_early_war_air_strategy days = 90 }
		}
	}
	
	abort_when_not_enabled = yes
	
	# Prioritize air superiority to gain control of airspace first
	ai_strategy = {
		type = mission_threshold
		id = air_superiority
		value = -200
	}
	
	# Keep reconnaissance high to spot enemy movements
	ai_strategy = {
		type = mission_threshold
		id = air_superiority
		value = -100
	}
	
	# Moderate focus on CAS to support ground operations
	ai_strategy = {
		type = mission_threshold
		id = cas
		value = -100
	}
	
	# Lower priority for strategic bombing early on
	ai_strategy = {
		type = mission_threshold
		id = strategic_bombing
		value = 100
	}
}

# Mid-war phase - balanced approach with emphasis on ground support
EAI_AIR_mid_war_mission_priority = {
	enable = {
		has_war = yes
		OR = {
			AND = {
				date > 1940.1.1
				date < 1943.1.1
				has_war_with = FROM
				NOT = { has_country_flag = EAI_late_war }
				NOT = { has_country_flag = EAI_early_war_air_strategy }
			}
			has_country_flag = { flag = EAI_mid_war_air_strategy days = 90 }
		}
	}
	
	abort_when_not_enabled = yes
	
	# Balance between air superiority and close air support
	ai_strategy = {
		type = mission_threshold
		id = air_superiority
		value = -150
	}
	
	# Increase CAS for ground support
	ai_strategy = {
		type = mission_threshold
		id = cas
		value = -150
	}
	
	# Start selective strategic bombing
	ai_strategy = {
		type = mission_threshold
		id = strategic_bombing
		value = -50
	}
	
	# More naval bombing if at war with naval powers
	ai_strategy = {
		type = mission_threshold
		id = naval_bombing
		value = -100
		target_country_trigger = {
			FROM = {
				OR = {
					tag = ENG
					tag = USA
					tag = JAP
					has_navy_size = { size > 49 }
				}
			}
		}
	}
}

### Theater-Based Air Wing Management ###

# Distribute air wings based on frontline importance
EAI_AIR_theater_distribution = {
	enable = {
		has_war = yes
		has_war_with = FROM
	}
	
	abort_when_not_enabled = yes
	
	# Prioritize air regions covering active fronts
	ai_strategy = {
		type = strategic_air_importance
		value = 200
		
		id = "REGION"
		
		region_trigger = {
			any_neighbor_state = {
				is_owned_by = ROOT
				is_controlled_by = ROOT
				is_fully_controlled_by = ROOT
				custom_trigger_tooltip = {
					tooltip = is_border_state
					is_border_state = yes
				}
			}
		}
	}
	
	# Even higher priority for regions with active combat
	ai_strategy = {
		type = strategic_air_importance
		value = 300
		
		id = "REGION"
		
		region_trigger = {
			any_neighbor_state = {
				is_owned_by = ROOT
				has_active_combat = yes
			}
		}
	}
	
	# Prioritize regions with enemy bombers active
	ai_strategy = {
		type = strategic_air_importance
		value = 250
		
		id = "REGION"
		
		region_trigger = {
			any_neighbor_state = {
				is_owned_by = ROOT
				FROM = { has_strategic_bombing_in_state = PREV }
			}
		}
	}
}

				is_core_of = ROOT

					has_strategic_bombing_in_state = PREV

	}
}

### Airbase Management ###

# Prevent overcrowding of airbases by redistributing planes
EAI_AIR_airbase_management = {
	enable = {
		has_war = yes
	}
	
	abort_when_not_enabled = yes
	
	# Increase priority for using alternative airbases
	ai_strategy = {
		type = air_base_use_percentage
		value = 80
	}
	
	# Give higher weight to dispersed air forces
	ai_strategy = {
		type = air_base_weight
		id = "FACTOR"
		value = -50
		
		air_base_weight_factor_trigger = {
			has_air_base_capacity_percentage = {
				tag = ROOT
				ratio > 0.8
			}
		}
	}
	
	# Evacuate air wings from threatened airbases
	ai_strategy = {
		type = air_base_preference
		id = "STATE"
		value = -100
		
		state_trigger = {
			is_owned_by = ROOT
			any_enemy_country = {
				has_war_with = ROOT
				controls_state = PREV
			}
		}
	}
}
