# This system enhances Japan's offensive capabilities against China
# to make them more aggressive in pushing inland instead of stalling at the frontline

# Basic strategy - higher aggression and area priorities
EAI_JAP_china_aggressive_stance = {
	enable = {
		tag = JAP
		is_ai = yes
		has_war = yes
		has_war_with = CHI
	}
	
	abort_when_not_enabled = yes
	
	# Make AI much more aggressive
	ai_strategy = {
		type = aggression
		value = 200  # Significantly higher than default
	}
	
	# Higher line pressure - push harder
	ai_strategy = {
		type = front_line_pressure
		value = 80   # Much higher than default (usually around 25-30)
	}
	
	# Don't be too cautious about attack odds
	ai_strategy = {
		type = enemy_strength_factor
		value = -20  # Less concerned about enemy strength
	}
}
